cmake_minimum_required(VERSION 3.24)
project(CoastQt6 LANGUAGES CXX)
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Core Gui Widgets)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
qt_add_plugin(coast_qpa STATIC CLASS_NAME CoastIntegrationPlugin PLUGIN_TYPE platforms coast.cpp)
target_link_libraries(coast_qpa PRIVATE Qt6::CorePrivate Qt6::GuiPrivate)
target_compile_options(coast_qpa PRIVATE -idirafter${CMAKE_CURRENT_SOURCE_DIR}/../../include)
target_compile_definitions(coast_qpa PRIVATE _LARGEFILE64_SOURCE _LARGEFILE_SOURCE)
qt_add_executable(qt6-demo demo.cpp)
target_link_libraries(qt6-demo PRIVATE coast_qpa Qt6::Widgets)
qt_import_plugins(qt6-demo NO_DEFAULT)
target_link_options(qt6-demo PRIVATE -static -no-pie -Wl,--gc-sections)
